{{/* Responsive WebP from assets/. Args: src, alt, widths, sizes, class, eager. */}} {{- $img := resources.Get .src -}} {{- if not $img }}{{ errorf "img partial: %q not found in assets" .src }}{{ end -}} {{- $ws := .widths | default (slice 480 800 1200) -}} {{- $srcs := slice -}} {{- $big := $img -}} {{- range $ws }} {{- if le . $img.Width }} {{- $r := $img.Resize (printf "%dx webp q80" .) }} {{- $srcs = $srcs | append (printf "%s %dw" $r.RelPermalink $r.Width) }} {{- $big = $r }} {{- end }} {{- end -}} {{- if not $srcs }}{{ $big = $img.Process "webp q80" }}{{ $srcs = slice (printf "%s %dw" $big.RelPermalink $big.Width) }}{{ end -}}
